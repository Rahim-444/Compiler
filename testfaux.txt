MainPrgm test_all_errors; < !- function name too long - !>
Var     
    < !- Lexical errors - !>     
    let very_long_identifier_name_error: Int;  < !- Identifier too long (>14 characters) - !>     
    let invalid_name_: Int;                    < !- Identifier ending with underscore - !>     
    let invalid__name: Int;                    < !- Identifiers with consecutive underscores - !>     
    @define Const TOO_BIG: Int = 40000;        < !- Integer out of range (>32767) - !>     
    @define Const TOO_SMALL: Int = (-40000);   < !- Integer out of range (<-32768) - !>     
    let $invalid: Int;                         < !- Unauthorized special character - !>      

    < !- Valid declarations to test other errors - !>     
    let x, y: Int;     
    let a, b: Float;     
    let arr: [Int; 5];     
    @define Const PI: Float = 3.14;     
    @define Const MAX: Int = 100;          < !- Double declaration error - !>     
    let x: Float;                              < !- Redeclaration of x already declared - !>  

BeginPg {
    < !- Syntax errors - !>     
    x := 10                                    < !- Missing semicolon - !>          
    x := 50 / (7 - 7);                         < !-  division by zero - !>          

    if x > 5 {                                 < !- Missing parentheses and 'then' keyword - !>         
        y := 20;     
    }          

    x := ++ 5;                                 < !- Incorrect operator usage - !>          

    x := (10 + 5;                              < !- Unbalanced parentheses - !>      

    < !- Semantic errors - Undeclared variables - !>     
    undeclared_var := 5;                       < !- Usage of undeclared variable - !>          

    < !- Semantic errors - Types - !>     
    x := "chaîne de caractères";               < !- Type mismatch (string to int) - !>          

    < !- Semantic errors - Constants - !>     
    PI := 3.14159;                             < !- Attempt to modify a constant - !>          

    < !- Semantic errors - Arrays - !>     
    y := arr;                                  < !- Using array without index - !>     
    z := x[0];                                 < !- Scalar used as array - !>     
    arr[a] := 10;                              < !- Non-integer index for array - !>     
    arr[10] := 5;                              < !- Array index out of bounds - !>          

    < !- Semantic errors - Division by zero - !>     
    x := 100 / 0;                              < !- Explicit division by zero - !>     
    y := 10;     

    < !- Semantic errors - Control structures - !>     
    if (a) then {                              < !- Non-boolean condition in if - !>         
        x := 10;     
    }          

    < !- Semantic errors - Loops - !>     
    for i from 1 to 10 step 0 {                < !- Step equal to zero - !>         
        x := i;     
    }          

    for PI from 1 to 10 step 1 {               < !- Constant used as loop variable - !>         
        x := 5;     
    }          

    for arr from 1 to 10 step 1 {              < !- Array used as loop variable - !>         
        x := 5;     
    }          

    for k from 1 to 10 step 1 {                < !- Undeclared loop variable - !>         
        x := k;     
    }          

    for i from 1.5 to 10 step 1 {              < !- Non-integer 'from' expression - !>         
        x := i;     
    }          

    for i from 1 to 10.5 step 1 {              < !- Non-integer 'to' expression - !>         
        x := i;     
    }          

    for i from 1 to 10 step 1.5 {              < !- Non-integer 'step' expression - !>         
        x := i;     
    }          

    < !- Semantic errors - Input/Output - !>     
    input(PI);                                 < !- Input with a constant - !>     
    input(arr);                                < !- Input with array without index - !>     
    input(arr[10]);                            < !- Input with out-of-bounds index - !>     
    output("Valeur de z: ", z);                < !- Output with undeclared identifier - !>          

    < !- Multiple errors in a complex expression - !>     
    x := (undeclared_var + PI[0]) / (MAX - MAX);  < !- Undeclared variable, constant used as array, division by zero - !>          

    < !- Logical condition error - !>     
    if (x > 10 AND y) then {                   < !- Right operand of AND is not a condition - !>         
        x := 5;     
    }          

    < !- Do-while loop with non-boolean condition - !>     
    do {         
        x := x - 1;     
    } while (a);                               < !- Non-boolean condition in while - !> 
} 
EndPg;
